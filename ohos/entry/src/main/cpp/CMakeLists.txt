cmake_minimum_required(VERSION 3.16)
project(modu_embedding LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT OHOS_ARCH STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Modu local embedding requires the audited OHOS arm64 runtime")
endif()
get_filename_component(MODU_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(MODU_ORT "${MODU_ROOT}/build/ohos-embedding/onnxruntime")
if(NOT EXISTS "${MODU_ORT}/include/onnxruntime_cxx_api.h")
  message(FATAL_ERROR "Run python3 scripts/harmony/prepare_embedding_runtime.py before building HAP")
endif()
add_library(onnxruntime SHARED IMPORTED)
set_target_properties(onnxruntime PROPERTIES IMPORTED_LOCATION "${MODU_ORT}/lib/libonnxruntime.so")
add_library(modu_embedding SHARED embedding_napi.cpp)
target_include_directories(modu_embedding PRIVATE "${MODU_ORT}/include")
target_link_libraries(modu_embedding PRIVATE onnxruntime libace_napi.z.so)
target_compile_options(modu_embedding PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter -Wno-unused-command-line-argument)
target_link_options(modu_embedding PRIVATE -Wl,--no-undefined -Wl,--no-allow-shlib-undefined)
set_target_properties(modu_embedding PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
